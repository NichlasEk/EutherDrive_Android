cmake_minimum_required(VERSION 3.16)
project(euther_n64_gpu LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(CMAKE_CXX_VISIBILITY_PRESET hidden)
set(CMAKE_C_VISIBILITY_PRESET hidden)
set(CMAKE_VISIBILITY_INLINES_HIDDEN ON)
set(PARALLEL_RDP_SOURCE "" CACHE PATH "Pinned paraLLEl-RDP checkout")
if (NOT EXISTS "${PARALLEL_RDP_SOURCE}/parallel-rdp/rdp_device.hpp")
    message(FATAL_ERROR "Set PARALLEL_RDP_SOURCE to the prepared checkout")
endif()
set(GRANITE_RENDERER OFF CACHE BOOL "" FORCE)
set(GRANITE_TOOLS OFF CACHE BOOL "" FORCE)
set(GRANITE_AUDIO OFF CACHE BOOL "" FORCE)
set(GRANITE_PLATFORM headless CACHE STRING "" FORCE)
set(GRANITE_VULKAN_FOSSILIZE OFF CACHE BOOL "" FORCE)
set(GRANITE_POSITION_INDEPENDENT ON CACHE BOOL "" FORCE)
add_subdirectory("${PARALLEL_RDP_SOURCE}" upstream EXCLUDE_FROM_ALL)
target_compile_definitions(granite-vulkan PUBLIC VULKAN_DEBUG)
target_compile_options(SPIRV PRIVATE -include cstdint)
target_compile_options(granite-util PUBLIC "$<$<COMPILE_LANGUAGE:CXX>:-include>" "$<$<COMPILE_LANGUAGE:CXX>:cstdint>")

# This dependency graph deliberately excludes alp-core and rdp-utils.
add_library(euther_n64_gpu SHARED n64_gpu.cpp)
target_link_libraries(euther_n64_gpu PRIVATE parallel-rdp granite-application-global-init granite-threading)
granite_setup_default_link_libraries(euther_n64_gpu)
target_compile_options(euther_n64_gpu PRIVATE -Wall -Wextra -Werror)
target_link_options(euther_n64_gpu PRIVATE
    "-Wl,--version-script=${CMAKE_CURRENT_SOURCE_DIR}/exports.map" "-Wl,--exclude-libs,ALL" "-Wl,-z,defs")
set_target_properties(euther_n64_gpu PROPERTIES VERSION 1.0.0 SOVERSION 1
    LINK_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/exports.map")
